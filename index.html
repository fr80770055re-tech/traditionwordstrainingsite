function doGet(e) { return ContentService.createTextOutput(JSON.stringify({"status": "success", "message": "API is working"})).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); // 🛡️ 防呆機制：如果找不到分頁，就自動幫老師建立 var sheetRecords = ss.getSheetByName("Records"); if (!sheetRecords) { sheetRecords = ss.insertSheet("Records"); } var sheetInventory = ss.getSheetByName("Inventory"); if (!sheetInventory) { sheetInventory = ss.insertSheet("Inventory"); sheetInventory.appendRow(["座號", "原木", "水滴石", "火焰碎片"]); } // 檢查字庫分頁是否存在（可能叫 WordPool 或 工作表1） var sheetWordPool = ss.getSheetByName("WordPool"); if (!sheetWordPool) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "找不到名為 WordPool 的分頁，請確認試算表下方的分頁名稱！"})).setMimeType(ContentService.MimeType.JSON); } // --- 讀取邏輯 --- if (action === "load") { var seat = data.seat.toString(); var materials = { wood: 0, water: 0, fire: 0 }; var lastRowInv = sheetInventory.getLastRow(); if (lastRowInv > 0) { var dataRange = sheetInventory.getDataRange().getValues(); for (var i = 1; i < dataRange.length; i++) { if (dataRange[i][0].toString() === seat) { materials = { wood: Number(dataRange[i][1])||0, water: Number(dataRange[i][2])||0, fire: Number(dataRange[i][3])||0 }; break; } } } var cloudWords = []; var wordValues = sheetWordPool.getDataRange().getValues(); var validRows = wordValues.slice(1).filter(function(r) { return r[0] && r[1] && r[2]; }); var shuffled = validRows.sort(function() { return 0.5 - Math.random(); }); var sampleSize = Math.min(15, shuffled.length); for(var j = 0; j < sampleSize; j++) { cloudWords.push({ word: shuffled[j][0].toString(), zhuyin: shuffled[j][1].toString(), example1: shuffled[j][2].toString(), example2: shuffled[j][3] ? shuffled[j][3].toString() : "", definition: shuffled[j][4] ? shuffled[j][4].toString() : "暫無注釋", flowerCore: shuffled[j][5] ? shuffled[j][5].toString() : " ", flowerPetals: shuffled[j][6] ? shuffled[j][6].toString() : "" }); } return ContentService.createTextOutput(JSON.stringify({"status": "success", "materials": materials, "words": cloudWords})) .setMimeType(ContentService.MimeType.JSON); } // --- 儲存邏輯 --- if (action === "save") { sheetRecords.appendRow([new Date(), data.grade, data.seat, data.timeSpent, data.errorCount]); var seat = data.seat.toString(); var added = data.addedMaterials; var lastRowInv = sheetInventory.getLastRow(); var foundRow = -1; if (lastRowInv > 0) { var dataRange = sheetInventory.getDataRange().getValues(); for (var i = 1; i < dataRange.length; i++) { if (dataRange[i][0].toString() === seat) { foundRow = i + 1; break; } } } if (foundRow !== -1) { sheetInventory.getRange(foundRow, 2).setValue(Number(sheetInventory.getRange(foundRow, 2).getValue()||0) + added.wood); sheetInventory.getRange(foundRow, 3).setValue(Number(sheetInventory.getRange(foundRow, 3).getValue()||0) + added.water); sheetInventory.getRange(foundRow, 4).setValue(Number(sheetInventory.getRange(foundRow, 4).getValue()||0) + added.fire); } else { sheetInventory.appendRow([seat, added.wood, added.water, added.fire]); } return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()})).setMimeType(ContentService.MimeType.JSON); } }
